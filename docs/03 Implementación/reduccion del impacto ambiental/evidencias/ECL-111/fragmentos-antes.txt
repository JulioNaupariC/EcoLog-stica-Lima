function AppRoutes() {
  const [identity, setIdentity] = useState<LoginResponse | null>(recallDriver)
  // Shown in the top bar only; kept in memory, never stored on the device.
  const [accountEmail, setAccountEmail] = useState('')
  const [logoutMessage, setLogoutMessage] = useState('')
  const [loggingOut, setLoggingOut] = useState(false)
  const navigate = useNavigate()
  const location = useLocation()
  const { notify } = useToast()
  const createOrdersAllowed = identity !== null && canManageOperations(identity.rol)
  const vehiclesAllowed = identity !== null && canListVehicles(identity.rol)
  const mustChangePassword = identity?.debe_cambiar_contrasena === true
  const signedIn = identity !== null
  // WCAG 2.4.2: the tab identifies the screen, including denied and public routes.
  useEffect(() => {
    const heading = document.querySelector('main h1')?.textContent?.trim()
    document.title = heading ? `${heading} — EcoLogística Lima` : 'EcoLogística Lima'
  }, [location.pathname, identity?.rol, mustChangePassword])
  useIncidentSynchronization(identity?.rol === 'CONDUCTOR' && !mustChangePassword ? identity.usuario_id : undefined)

  function handleLoginSuccess(loggedInIdentity: LoginResponse, email: string) {
    // A temporary-password session is not remembered as a usable driver tab.
    if (!loggedInIdentity.debe_cambiar_contrasena) rememberDriver(loggedInIdentity)
    setIdentity(loggedInIdentity)
    setAccountEmail(email)
    setLogoutMessage('')
    void navigate(loggedInIdentity.debe_cambiar_contrasena ? CHANGE_PASSWORD_PATH : '/', { replace: true })
  }

  // ECL-77: any API 403 "Cambio de contraseña requerido" locks the app on the change page.
  useEffect(() => {
    if (!signedIn) return undefined
    return installPasswordChangeGuard(() => {
      forgetDriver()
      setIdentity(current => current ? { ...current, debe_cambiar_contrasena: true } : current)
      void navigate(CHANGE_PASSWORD_PATH, { replace: true })
    })
  }, [signedIn, navigate])

  function handlePasswordChanged() {
    if (identity) {
      const updated: LoginResponse = { usuario_id: identity.usuario_id, rol: identity.rol,
        ...(identity.expires_at ? { expires_at: identity.expires_at } : {}) }
      rememberDriver(updated)
      setIdentity(updated)
    }
    void navigate('/', { replace: true })
    notify('Contraseña actualizada.')
  }

  const handleSessionExpired = useCallback(() => {
    forgetDriver()
    setIdentity(null)
    setLogoutMessage('Tu sesión ha vencido. Vuelve a iniciar sesión.')
    void navigate('/login', { replace: true })
  }, [navigate])

  useEffect(() => {
    if (!identity?.expires_at) return
    const remaining = Date.parse(identity.expires_at) - Date.now()
    const timer = setTimeout(handleSessionExpired, Math.max(0, remaining))
    return () => clearTimeout(timer)
  }, [identity, handleSessionExpired])

  async function handleLogout() {
    if (!identity || loggingOut) return
    setLogoutMessage('')
    setLoggingOut(true)
    try {
      if (identity.rol === 'CONDUCTOR') {
        let pendingCount: number | null
        try {
          pendingCount = (await listPendingReports(identity.usuario_id)).length +
            (await listIncidents(identity.usuario_id)).filter(item => !item.acknowledgement).length
        } catch {
          pendingCount = null
        }
        if (pendingCount === null) {
          if (!window.confirm(
            'No se pudo verificar la cola offline. Los reportes locales no se eliminarán. ¿Cerrar sesión?',
          )) return
        } else if (pendingCount > 0 && !window.confirm(
          `Tienes ${pendingCount === 1 ? '1 reporte pendiente' : `${pendingCount} reportes pendientes`}. Se conservarán en este dispositivo y podrás sincronizarlos al volver a iniciar sesión. ¿Cerrar sesión?`,
        )) {
          return
        }
      }
      await logout()
      forgetDriver()
      setIdentity(null)
      setAccountEmail('')
      await navigate('/login', { replace: true })
      notify('Sesión cerrada.')
      if (identity.rol === 'CONDUCTOR') {
        try {
          await clearItinerary(identity.usuario_id)
        } catch {
          setLogoutMessage('La sesión se cerró, pero no se pudo borrar el itinerario guardado en este dispositivo.')
        }
      }
    } catch (error) {
      setLogoutMessage(
        error instanceof Error ? error.message : 'No se pudo cerrar la sesión.',
      )
    } finally {
      setLoggingOut(false)
    }
  }

  if (identity === null) {
    return <PublicShell>
      <Routes>
        <Route path="/login" element={<LoginPage onLoginSuccess={handleLoginSuccess} notice={logoutMessage} />} />
        <Route path="*" element={<Navigate to="/login" replace />} />
      </Routes>
    </PublicShell>
  }

  const banner = logoutMessage
    ? <div className="shell-banner"><Alert tone="danger">{logoutMessage}</Alert></div>
    : null
  if (mustChangePassword) {
    return <PublicShell>
      {banner}
      <Routes>
        <Route path={CHANGE_PASSWORD_PATH} element={<ChangePasswordPage required onChanged={handlePasswordChanged}
          onSessionExpired={handleSessionExpired} onLogout={() => { void handleLogout() }} />} />
        <Route path="*" element={<Navigate to={CHANGE_PASSWORD_PATH} replace />} />
      </Routes>
    </PublicShell>
  }
  const guarded = (allowed: boolean, element: ReactNode) => allowed ? element : <AccessDenied />
  const shellProps = {
    role: identity.rol,
    accountName: accountEmail || ROLE_LABEL[identity.rol],
    loggingOut,
    onLogout: () => { void handleLogout() },
    banner,
  }
  const routes = <Routes>
    <Route path="/" element={<HomePage role={identity.rol} ownerId={identity.usuario_id} />} />
    <Route path="/login" element={<Navigate to="/" replace />} />
    <Route path="/pedidos/nuevo" element={guarded(createOrdersAllowed, <OrderCreatePage />)} />
    <Route path="/operacion/incidencias" element={guarded(createOrdersAllowed,
      <OperationIncidentsPage key={identity.usuario_id} onSessionExpired={handleSessionExpired} />)} />
    <Route path="/vehiculos" element={guarded(vehiclesAllowed, <VehiclesPage canCreate={createOrdersAllowed} />)} />
    <Route path="/conductor/itinerario" element={guarded(identity.rol === 'CONDUCTOR',
      <DriverItineraryPage key={identity.usuario_id} ownerId={identity.usuario_id} onSessionExpired={handleSessionExpired} />)} />
    <Route path="/conductor/incidencias/nueva" element={guarded(identity.rol === 'CONDUCTOR',
      <IncidentCreatePage key={identity.usuario_id} ownerId={identity.usuario_id} />)} />
    <Route path="/conductores" element={guarded(createOrdersAllowed,
      <DriversPage key={identity.usuario_id} onSessionExpired={handleSessionExpired} />)} />
    <Route path="/clientes/preferencias" element={guarded(createOrdersAllowed,
      <DeliveryPreferencesPage key={identity.usuario_id} onSessionExpired={handleSessionExpired} />)} />
    <Route path="/usuarios" element={guarded(canManageUsers(identity.rol),
      <UsersPage key={identity.usuario_id} ownId={identity.usuario_id} onSessionExpired={handleSessionExpired} />)} />
    <Route path={CHANGE_PASSWORD_PATH} element={<ChangePasswordPage required={false}
      onChanged={handlePasswordChanged} onSessionExpired={handleSessionExpired} />} />
    <Route path="*" element={<NotFoundPage />} />
  </Routes>
  return identity.rol === 'CONDUCTOR'
    ? <DriverShell {...shellProps}>{routes}</DriverShell>
    : <StaffShell {...shellProps}>{routes}</StaffShell>
}

type VehicleField = keyof VehicleCreatePayload
type FormValues = Record<VehicleField, string>
type FieldErrors = Partial<Record<VehicleField, string>>

const INITIAL_VALUES: FormValues = {

  tipo: '',

const FIELDS: { name: VehicleField; label: string; hint: string }[] = [

const TYPE_LABEL: Record<string, string> = { CAMIONETA: 'Camioneta', FURGON: 'Furgón', MOTO: 'Moto' }

const DECIMAL_FIELDS = [
  { name: 'capacidad_kg', places: 2, allowZero: false },
  { name: 'capacidad_m3', places: 2, allowZero: false },
  { name: 'rendimiento_km_l', places: 3, allowZero: false },
  { name: 'factor_co2_kg_km', places: 5, allowZero: true },
] as const

function isVehicleType(value: string): value is VehicleType {
  return value === 'CAMIONETA' || value === 'FURGON' || value === 'MOTO'
}

function decimalError(value: string, places: number, allowZero: boolean): string | undefined {
  if (!value.trim()) return 'Este campo es obligatorio.'
  if (!/^\d+(?:\.\d+)?$/u.test(value)) {
    return 'Usa un número sin signo, con punto decimal; por ejemplo, 1.25.'
  }
  const [whole, fraction = ''] = value.split('.')
  // Count on a copy, preserving the original text and avoiding floating point.
  const integerDigits = whole.replace(/^0+/u, '')
  const fractionalDigits = fraction.replace(/0+$/u, '')
  if (!allowZero && !/[1-9]/u.test(value)) return 'El valor debe ser mayor que cero.'
  if (integerDigits.length > 10 - places || fractionalDigits.length > places) {
    return `Usa como máximo ${10 - places} dígitos enteros y ${places} decimales significativos.`
  }
  return undefined
}

function validate(values: FormValues): { errors: FieldErrors; payload?: VehicleCreatePayload } {
  const errors: FieldErrors = {}
  const plateLength = Array.from(values.placa.trim().toUpperCase()).length
  if (plateLength === 0) errors.placa = 'La placa es obligatoria.'
  else if (plateLength > 10) errors.placa = 'La placa no puede superar 10 caracteres al normalizarse.'
  if (!isVehicleType(values.tipo)) errors.tipo = 'Selecciona un tipo de vehículo válido.'
  for (const { name, places, allowZero } of DECIMAL_FIELDS) {
    const error = decimalError(values[name], places, allowZero)
    if (error) errors[name] = error
  }
  const year = Number(values.anio_fabricacion)
  if (!/^\d{4}$/u.test(values.anio_fabricacion) || year < 1980 || year > 2100) {
    errors.anio_fabricacion = 'Ingresa un año entero entre 1980 y 2100.'
  }
  if (Object.keys(errors).length > 0 || !isVehicleType(values.tipo)) return { errors }
  return { errors, payload: { ...values, tipo: values.tipo, anio_fabricacion: year } }
}

function serviceMessage(error: unknown): string {
  return error instanceof VehicleServiceError ? error.message : 'No se pudo completar la operación.'
}

function VehicleForm({ onCreated }: { onCreated: (vehicle: VehicleResponse) => void }) {
  const [values, setValues] = useState<FormValues>(INITIAL_VALUES)
  const [errors, setErrors] = useState<FieldErrors>({})
  const [globalError, setGlobalError] = useState('')
  const [submitting, setSubmitting] = useState(false)
  const submittingRef = useRef(false)
  const mountedRef = useRef(false)
  const formRef = useRef<HTMLFormElement>(null)
  const errorRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    mountedRef.current = true
    return () => { mountedRef.current = false }
  }, [])

  useEffect(() => {
    if (!globalError || submitting) return
    const firstInvalid = FIELDS.find(({ name }) => errors[name])
    const control = firstInvalid && formRef.current?.elements.namedItem(firstInvalid.name)
    if (control instanceof HTMLElement) control.focus()
    else errorRef.current?.focus()
  }, [errors, globalError, submitting])

  function updateValue(field: VehicleField, value: string) {
    setValues((current) => ({ ...current, [field]: value }))
    setErrors((current) => ({ ...current, [field]: undefined }))
    setGlobalError('')
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    if (submittingRef.current) return
    const result = validate(values)
    setErrors(result.errors)
    if (!result.payload) {
      setGlobalError('Revisa los campos indicados antes de registrar el vehículo.')
      return
    }

    submittingRef.current = true
    setSubmitting(true)
    setGlobalError('')
    try {
      const vehicle = await createVehicle(result.payload)
      if (!mountedRef.current) return
      setValues(INITIAL_VALUES)
      setErrors({})
      onCreated(vehicle)
    } catch (error) {
      if (!mountedRef.current) return
      const nextErrors: FieldErrors = {}
      if (error instanceof VehicleServiceError) {
        for (const issue of error.issues) nextErrors[issue.field] = issue.message
        if (error.kind === 'conflict') nextErrors.placa = error.message
      }
      setErrors(nextErrors)
      setGlobalError(serviceMessage(error))
    } finally {
      submittingRef.current = false
      if (mountedRef.current) setSubmitting(false)
    }
  }

  function renderField(name: VehicleField) {
    const field = FIELDS.find(item => item.name === name)!
    const id = `vehicle-${name}`
    const describedBy = [errors[name] ? `${id}-error` : null, `${id}-hint`].filter(Boolean).join(' ')
    const inputProps = {
      id,
      name,
      required: true,
      disabled: submitting,
      value: values[name],
      'aria-invalid': Boolean(errors[name]),
      'aria-describedby': describedBy,
    }
    return (
      <div className="form-field" key={name}>
        <label htmlFor={id}>{field.label}</label>
        {name === 'tipo' ? (
          <select {...inputProps} onChange={(event) => updateValue(name, event.target.value)}>
            <option value="">Selecciona un tipo</option>
            <option value="CAMIONETA">Camioneta</option>
            <option value="FURGON">Furgón</option>
            <option value="MOTO">Moto</option>
          </select>
        ) : (
          <input
            {...inputProps}
            type="text"
            inputMode={name === 'placa' ? 'text' : name === 'anio_fabricacion' ? 'numeric' : 'decimal'}
            onChange={(event) => updateValue(name, event.target.value)}
          />
        )}
        {errors[name] ? <p className="field-error" id={`${id}-error`}>{errors[name]}</p> : null}
        <p className="field-hint" id={`${id}-hint`}>{field.hint}</p>
      </div>
    )
  }

  return (
    <Card title="Registrar vehículo" titleId="vehicle-form-title" className="vehicle-registration">
      <p id="vehicle-format-help" className="card-description">
        Todos los campos son obligatorios. Usa punto decimal, sin separadores de miles ni exponentes.
      </p>
      {globalError ? (
        <Alert tone="danger" ref={errorRef} tabIndex={-1}>{globalError}</Alert>
      ) : null}
      {submitting ? <p className="field-hint" role="status">Registrando vehículo…</p> : null}
      <form
        className="form-stack"
        aria-labelledby="vehicle-form-title"
        aria-describedby="vehicle-format-help"
        aria-busy={submitting}
        noValidate
        ref={formRef}
        onSubmit={(event) => { void handleSubmit(event) }}
      >
        {SECTIONS.map(section => (
          <FormSection key={section.legend} legend={section.legend} description={section.description}>
            <div className="form-grid">{section.fields.map(renderField)}</div>
          </FormSection>
        ))}
        <FormActions>
          <Button type="submit" icon={Plus} disabled={submitting}>
            {submitting ? 'Registrando…' : 'Registrar vehículo'}
          </Button>
        </FormActions>
      </form>
    </Card>
  )
}

interface ListState {
  status: 'loading' | 'ready' | 'error'
  vehicles: VehicleResponse[]
  error: string
}

export function VehiclesPage({ canCreate }: { canCreate: boolean }) {
  const [list, setList] = useState<ListState>({ status: 'loading', vehicles: [], error: '' })
  const [createdVehicle, setCreatedVehicle] = useState<VehicleResponse | null>(null)
  const latestRequest = useRef(0)
  const listErrorRef = useRef<HTMLDivElement>(null)

  const loadList = useCallback((afterCreate = false) => {
    const requestId = ++latestRequest.current
    void listVehicles().then(
      (vehicles) => {
        if (requestId !== latestRequest.current) return
        setList({ status: 'ready', vehicles, error: '' })
      },
      (error: unknown) => {
        if (requestId !== latestRequest.current) return
        const message = serviceMessage(error)
        setList((current) => ({
          ...current,
          status: 'error',
          error: afterCreate ? `Vehículo registrado. No se pudo actualizar el listado. ${message}` : message,
        }))
      },
    )
  }, [])

  useEffect(() => {
    loadList()
    // Invalidates earlier GETs, including StrictMode's first mount and unmount.
    return () => { latestRequest.current += 1 }
  }, [loadList])

  useEffect(() => {
    if (list.status === 'error') listErrorRef.current?.focus()
  }, [list])

  function refreshList(afterCreate = false) {
    setList((current) => ({ ...current, status: 'loading', error: '' }))
    loadList(afterCreate)
  }

  function handleCreated(vehicle: VehicleResponse) {
    setCreatedVehicle(vehicle)
    refreshList(true)
  }

  const active = list.vehicles.filter(vehicle => vehicle.estado === 'ACTIVO').length
  return (
    <section className="vehicles-page" aria-labelledby="vehicles-title">
      <PageHeader titleId="vehicles-title" title="Vehículos" eyebrow="Gestión de flota"
        breadcrumbs={[{ label: 'Inicio', to: '/' }, { label: 'Vehículos' }]}
        description="Unidades disponibles para la planificación, con su capacidad y factor de emisión." />
      {createdVehicle ? (
        <Alert tone="success" role="status">Vehículo registrado: {createdVehicle.placa}.</Alert>
      ) : null}
      {list.status === 'loading' ? <p className="visually-hidden" role="status">Cargando vehículos…</p> : null}
      <Card as="div" title="Listado de vehículos" titleId="vehicle-list-title" className="vehicle-list"
        description={list.vehicles.length ? `${list.vehicles.length} registrados · ${active} activos` : undefined}
        actions={<Button variant="secondary" size="sm" icon={RefreshCw} disabled={list.status === 'loading'} onClick={() => refreshList()}>
          Actualizar listado
        </Button>}>
        <div aria-busy={list.status === 'loading'} className="card-body" role="region" aria-label="Listado de vehículos">
          {list.status === 'loading' && list.vehicles.length === 0 ? <Skeleton lines={4} /> : null}
          {list.status === 'error' ? (
            <Alert tone="danger" ref={listErrorRef} tabIndex={-1}>
              {list.error}
              {list.vehicles.length > 0 ? <span> El listado mostrado puede estar desactualizado.</span> : null}
            </Alert>
          ) : null}
          {list.status === 'ready' && list.vehicles.length === 0 ? (
            <EmptyState icon={Truck} title="Sin vehículos" headingLevel={3}>
              <p role="status">No hay vehículos registrados.</p>
            </EmptyState>
          ) : null}
          {list.vehicles.length > 0 ? <VehicleTable vehicles={list.vehicles} /> : null}
        </div>
      </Card>
      {canCreate ? <VehicleForm onCreated={handleCreated} /> : null}
    </section>
  )
}

function availability(start: string | null, end: string | null): string {

export function DriversPage({ onSessionExpired }: { onSessionExpired: () => void }) {

  const [page, setPage] = useState(1)
  const [editor, setEditor] = useState<Editor>(null)
  const [opening, setOpening] = useState<string | null>(null)
  const [detailError, setDetailError] = useState('')
  const [saved, setSaved] = useState<DriverSummary | null>(null)
  const [success, setSuccess] = useState('')
  const [blocked, setBlocked] = useState(false)
  const requestNumber = useRef(0)
  const detailNumber = useRef(0)
  const listHeading = useRef<HTMLHeadingElement>(null)
  const accessError = useCallback((error: DriverServiceError) => {
    if (error.kind === 'unauthorized') onSessionExpired()
    if (error.kind === 'forbidden') {
      setBlocked(true); setEditor(null); setDetailError(error.message)
      setList({ status: 'error', result: null, error: error.message })
    }
  }, [onSessionExpired])
  const load = useCallback(async (nextPage: number) => {
    const number = ++requestNumber.current
    setPage(nextPage)

      const result = await listDrivers(nextPage)
      if (number === requestNumber.current) setList({ status: 'ready', result, error: '' })

      if (number !== requestNumber.current) return
      if (error instanceof DriverServiceError) accessError(error)
      setList({ status: 'error', result: null, error: error instanceof DriverServiceError ? error.message : 'No se pudo cargar el listado.' })
    }

    const lists = requestNumber

      if (error instanceof DriverServiceError) accessError(error)
      setList({ status: 'error', result: null, error: error instanceof DriverServiceError ? error.message : 'No se pudo cargar el listado.' })

  const [list, setList] = useState<ListState>({ status: 'loading', result: null, error: '' })
  const [page, setPage] = useState(1)
  const [step, setStep] = useState<Step>(null)
  const opener = useRef<HTMLButtonElement | null>(null)
  const cancelButton = useRef<HTMLButtonElement>(null)
  const doneButton = useRef<HTMLButtonElement>(null)
  const requestNumber = useRef(0)

  const load = useCallback(async (nextPage: number) => {
    const number = ++requestNumber.current
    setPage(nextPage)
    setList({ status: 'loading', result: null, error: '' })
    try {
      const result = await listUsers(nextPage, PAGE_SIZE)
      if (number === requestNumber.current) setList({ status: 'ready', result, error: '' })
    } catch (error) {
      if (number !== requestNumber.current) return
      if (error instanceof UserServiceError && error.kind === 'unauthorized') onSessionExpired()
      setList({ status: 'error', result: null,
        error: error instanceof UserServiceError ? error.message : 'No se pudo cargar el listado.' })
    }
  }, [onSessionExpired])

  useEffect(() => {
    const requests = requestNumber
    void Promise.resolve().then(() => load(1))
    return () => { requests.current++ }
  }, [load])

  async function confirmReset() {
    if (step?.kind !== 'confirm' || step.busy) return
    const { user } = step
    setStep({ ...step, busy: true, error: '' })
    try {
      const temporary = await resetPassword(user.usuario_id)
      setStep({ kind: 'show', user, temporary, copy: '' })
    } catch (error) {
      if (error instanceof UserServiceError && error.kind === 'unauthorized') { setStep(null); onSessionExpired(); return }
      const detail = error instanceof UserServiceError && ['not_found', 'conflict', 'forbidden'].includes(error.kind)
        ? error.message : RESET_FAILED
      setStep({ kind: 'confirm', user, busy: false, error: detail })
    }
  }

  async function copyTemporary() {
    if (step?.kind !== 'show') return
    try {
      await navigator.clipboard.writeText(step.temporary.contrasena_temporal)
      setStep({ ...step, copy: 'copied' })
    } catch {
      setStep({ ...step, copy: 'failed' })
    }
  }

  function finish() {
    if (step?.kind !== 'show') return
    const resetId = step.user.usuario_id
    // The temporary password leaves the component state here and is not kept anywhere.

function occurrence(value: string) {
  const parts = limaTime.formatToParts(new Date(value))
  const part = (type: Intl.DateTimeFormatPartTypes) => parts.find(item => item.type === type)?.value

}

/** ECL-75: read-only pending incidents for Administrator and Operator. */
export function OperationIncidentsPage({ onSessionExpired }: { onSessionExpired: () => void }) {
  const [list, setList] = useState<ListState>({ status: 'loading', page: 1 })
  const requests = useRef(0)
  const controller = useRef<AbortController | null>(null)

  const load = useCallback(async (page: number) => {
    const number = ++requests.current
    controller.current?.abort()
    const current = new AbortController()
    controller.current = current
    setList({ status: 'loading', page })
    try {
      let result = await listPendingIncidents(page, PAGE_SIZE, current.signal)
      if (number !== requests.current || current.signal.aborted) return
      // A concurrent validation can remove the last page. Return to a valid page.
      const lastPage = Math.max(1, Math.ceil(result.total / PAGE_SIZE))
      if (page > lastPage) result = await listPendingIncidents(lastPage, PAGE_SIZE, current.signal)
      if (number === requests.current && !current.signal.aborted) {
        setList({ status: 'ready', page: result.page, result })
      }
    } catch (error) {
      if (number !== requests.current || current.signal.aborted) return
      if (error instanceof IncidentListError && error.kind === 'unauthorized') { onSessionExpired(); return }
      setList({ status: 'error', page,
        message: error instanceof IncidentListError ? error.message : 'No se pudo cargar el listado. Intenta de nuevo.',
        forbidden: error instanceof IncidentListError && error.kind === 'forbidden' })
    }
  }, [onSessionExpired])

  useEffect(() => {
    let mounted = true
    void Promise.resolve().then(() => { if (mounted) void load(1) })
    const activeRequests = requests
    const activeController = controller
    return () => {
      mounted = false
      activeRequests.current++
      activeController.current?.abort()
    }
  }, [load])

  const busy = list.status === 'loading'
  const totalPages = list.status === 'ready' ? Math.max(1, Math.ceil(list.result.total / PAGE_SIZE)) : list.page

  return <section className="incidents-page" aria-labelledby="incidents-title">
    <PageHeader titleId="incidents-title" title="Incidencias pendientes" eyebrow="Operación"
      breadcrumbs={[{ label: 'Inicio', to: '/' }, { label: 'Incidencias pendientes' }]}
      description="Consulta los reportes de los conductores pendientes de validación, del más reciente al más antiguo."
      actions={<Button variant="secondary" icon={RefreshCw} disabled={busy} onClick={() => { void load(list.page) }}>
        Actualizar listado
      </Button>} />
    <section className="card" aria-labelledby="incident-list-title" aria-busy={busy}>
      <div className="card-header">
        <div className="card-heading">
          <h2 id="incident-list-title" className="card-title">Reportes por revisar</h2>
          <p className="card-description" role="status">
            {list.status === 'ready'
              ? `${list.result.total} ${list.result.total === 1 ? 'incidencia pendiente' : 'incidencias pendientes'} · Hora de Lima (UTC−5)`
              : 'Los reportes se muestran en hora de Lima (UTC−5).'}

      </> : null}
      <nav className="pagination" aria-label="Páginas de incidencias" hidden={list.status !== 'ready'}>
        <Button variant="secondary" size="sm" icon={ChevronLeft} disabled={busy || list.page <= 1}
          onClick={() => { void load(list.page - 1) }}>Anterior</Button>

    return new VehicleServiceError('conflict', { status })

    tipo: value.tipo as PendingIncident['tipo'], severidad: value.severidad as PendingIncident['severidad'],
    ocurrida_en: value.ocurrida_en, registrada_en: value.registrada_en,
    conductor: { conductor_id: value.conductor.conductor_id, nombre: value.conductor.nombre },
    latitud: value.latitud, longitud: value.longitud, observacion: value.observacion, estado: value.estado,
  }
}

/** ECL-73 / RN-022: server pagination, newest occurrence first, ID ascending on ties. */
export async function listPendingIncidents(page = 1, pageSize = 20, signal?: AbortSignal): Promise<IncidentPage> {
  if (!Number.isInteger(page) || page < 1 || !Number.isInteger(pageSize) || pageSize < 1 || pageSize > 100) {
    throw new IncidentListError('unexpected')
  }
  let url: string
  try {
    url = buildApiUrl(`incidencias?estado=PENDIENTE_VALIDACION&page=${page}&page_size=${pageSize}`)
  } catch { throw new IncidentListError('unexpected') }
  let response: Response
  try {
    const timeout = AbortSignal.timeout(15_000)
    response = await fetch(url, {
      credentials: 'include', cache: 'no-store', signal: signal ? AbortSignal.any([signal, timeout]) : timeout,
    })
  } catch (error) {
    if (signal?.aborted) throw error
    throw new IncidentListError('network')
  }
  if (!response.ok) {
    throw new IncidentListError(response.status === 401 ? 'unauthorized'
      : response.status === 403 ? 'forbidden' : response.status >= 500 ? 'unavailable' : 'unexpected')
  }
  if (response.status !== 200) throw new IncidentListError('unexpected')
  let body: unknown
  try { body = await response.json() } catch { throw new IncidentListError('unexpected') }
  if (!record(body) || body.page !== page || body.page_size !== pageSize ||
    typeof body.total !== 'number' || !Number.isSafeInteger(body.total) || body.total < 0 ||
    !Array.isArray(body.items) || body.items.length > pageSize || body.items.length > body.total) {
    throw new IncidentListError('unexpected')
  }
  const items = body.items.map(incident).sort((a, b) =>
    Date.parse(b.ocurrida_en) - Date.parse(a.ocurrida_en) || a.incidencia_id.localeCompare(b.incidencia_id))
  return { items, page, page_size: pageSize, total: body.total }
}
